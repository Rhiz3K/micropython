def item(data):
    return (len(data), hashlib.sha256(data).hexdigest())

def rows(files):
    return [dict(path=p, size=v[0], sha256=v[1]) for p, v in files.items()]

def mapping(data):
    return {r['path']: (r['size'], r['sha256']) for r in data}

def write_json(path, data):
    with Path(path).open('x') as f:
        json.dump(data, f)

class Tests(unittest.TestCase):

    def test_import_is_inert(self):
        fresh = importlib.util.module_from_spec(spec)
        with patch('builtins.open', side_effect=AssertionError('IO')), patch.object(Path, 'read_bytes', side_effect=AssertionError('IO')), patch.object(Path, 'read_text', side_effect=AssertionError('IO')):
            exec(compile(TEXT, str(SOURCE), 'exec'), fresh.__dict__)
        self.assertIsNone(fresh.p)
        self.assertIsNone(fresh.timing)

    def test_exact_probe_and_codec_gate(self):
        probe = good_probe()
        m.probe_pass(probe)
        for key, value in [('result', 'FAIL'), ('image_valid', False), ('format', 'OTHER'), ('decoder_sha256', 'f' * 64), ('red_plane_empty', False), ('plane_bytes', [4736]), ('width', 128), ('source_review_sha256', 'f' * 64), ('panel_refresh_performed', True)]:
            with self.subTest(key=key), self.assertRaises(RuntimeError):
                m.probe_pass(dict(probe, **{key: value}))

    def test_fresh_stage_snapshot_guard_before_dependencies_and_armed_last(self):
        for fail in (None, 'staging', 'after_park'):
            with self.subTest(fail=fail), tempfile.TemporaryDirectory() as d:
                root = Path(d)
                fake = Fake(root, fail)
                before = copy.deepcopy(fake.s)
                oldsha = m.sha(root / 'full-flash-before.bin')
                with patch.multiple(m, ROOT=root, OUT=root / 'pilot3', REVIEW=root / 'adapter-usb-v3-review', BINDING=root / 'adapter-usb-v3-review/dns-policy-binding-private.json', AUTHORIZED_WIFI=fake.auth, p=fake.p, timing=types.SimpleNamespace(atomic=fake.atomic), READONLY_DEPENDENCY_SHA=fake.dependencies, APPROVAL_SHA='d' * 64), patch.object(m, 'validate_approval', return_value=(fake.approval, {})):
                    if fail:
                        with self.assertRaises((RuntimeError, OSError)):
                            m.stage()
                        self.assertNotIn(m.ARMED, fake.files)
                        if fail == 'after_park':
                            self.assertNotIn('/main.py', fake.files)
                            self.assertEqual(fake.files[m.PARKED], fake.original['/main.py'])
                        else:
                            self.assertEqual(fake.files['/main.py'], fake.original['/main.py'])
                    else:
                        m.stage()
                        self.assertEqual(fake.trace[:4], ['get_original_main', ('put', m.TEMP), ('rename', '/main.py', m.PARKED), ('rename', m.TEMP, '/main.py')])
                        self.assertEqual([t for t in fake.trace if isinstance(t, tuple) and t[0] == 'atomic'][-1], ('atomic', m.ARMED))
                        self.assertIn('/app_image.py', fake.files)
                        self.assertEqual(fake.trace[-2:], ['friendly', 'close'])
                    saved = m.read_json(root / 'pilot3/fresh-initial-state-private.json')
                    self.assertEqual(saved['backup'], before['backup'])
                    self.assertEqual(saved['rtc'], before['rtc'])
                    self.assertEqual(saved['scope'], 'PILOT3_FRESH_ORIGINAL_BEFORE_STAGE')
                    self.assertEqual(fake.s, before)
                    self.assertEqual(m.sha(root / 'full-flash-before.bin'), oldsha)

    def test_postcheck_refuses_repl_without_complete_tail(self):
        for proof in ({'result': 'FAIL'}, {'result': 'BOUNDARIES_COMPLETE', 'completed_intervals': 3, 'boundaries': [{}, {}, {}, {'receipt_monotonic_ns': 1}], 'go_acknowledged': True, 'finished_monotonic_ns': 2}):
