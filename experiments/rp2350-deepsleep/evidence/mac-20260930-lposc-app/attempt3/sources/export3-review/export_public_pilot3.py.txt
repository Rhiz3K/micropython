def require(ok, reason):
    if not ok:
        raise ValueError(reason)

def regular(path, limit=2000000000):
    path = Path(path)
    require(path.is_file() and (not path.is_symlink()) and (path.stat().st_size <= limit), 'missing_or_invalid_regular_input')
    require(not any((p.is_symlink() for p in path.parents)), 'symlink_parent')
    return path

def sha(path):
    h = hashlib.sha256()
    with regular(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()

def canonical(obj):
    return json.dumps(obj, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()

def strict_json(content):

    def unique(pairs):
        value = {}
        for key, child in pairs:
            require(key not in value, 'duplicate_JSON_key')
            value[key] = child
        return value

    def invalid_constant(value):
        raise ValueError('nonfinite_JSON_constant')
    return json.loads(content, object_pairs_hook=unique, parse_constant=invalid_constant)

def object_hash(obj):
    return hashlib.sha256(canonical(obj)).hexdigest()

def read(path):
    value = strict_json(regular(path, 2000000).read_text())
    require(type(value) is dict, 'object_required')
    return value

def stamp(row):
    require(type(row['host_monotonic']) in (int, float) and math.isfinite(row['host_monotonic']) and (row['host_monotonic'] > 0), 'host_stamp')
    utc(row['host_utc'])

def utc(value):
    require(type(value) is str and re.fullmatch('\\d{4}-\\d\\d-\\d\\dT\\d\\d:\\d\\d:\\d\\d(?:\\.\\d+)?\\+00:00', value), 'UTC_stamp')
    dt.datetime.fromisoformat(value)

def finite_tree(value):
    if value is None or type(value) is bool or type(value) is int:
        return
    if type(value) is float:
        require(math.isfinite(value), 'nonfinite_value')
    elif type(value) is list:
        for item in value:
            finite_tree(item)
    elif type(value) is dict:
        for item in value.values():
            finite_tree(item)
    else:
        raise ValueError('unexpected_text_in_numeric_tree')

def stats(value):
    require(type(value) is dict and set(value) == STATS_KEYS and (value['quantiles_note'] == QUANTILES_NOTE), 'stats_schema')
    finite_tree({k: v for k, v in value.items() if k != 'quantiles_note'})

def validate_raw_row(kind, row, reader):
    if kind in ('frames', 'samples'):
        require(row.get('phase') in ('recording', 'drain'), 'raw_phase')
        utc(row['host_receipt_utc'])
        finite_tree({k: row[k] for k in COMMON_FRAME - {'phase', 'host_receipt_utc'}})
        if kind == 'samples':
            require(set(row) == SAMPLE_KEYS, 'sample_allowlist')
            finite_tree({k: row[k] for k in SAMPLE_KEYS - {'phase', 'host_receipt_utc'}})
        else:
            parsed = reader.parse_frame(bytes.fromhex(row['raw_hex']))
            require(parsed['valid'] is True and set(row) == set(parsed) - {'samples'} | COMMON_FRAME, 'raw_frame_allowlist_CRC')
            require(all((row[k] == v for k, v in parsed.items() if k != 'samples')), 'raw_frame_decoding')
    else:
        basic = {'event', 'host_receipt_utc', 'host_monotonic_ns'}
        utc(row['host_receipt_utc'])
        require(type(row['host_monotonic_ns']) is int, 'event_clock')
        extra = {'opened_identity_verified': set(), 'command_sent': {'command', 'report_hex', 'bytes_reported'}, 'recording_started': {'deadline_monotonic_ns'}, 'health': HEALTH_KEYS, 'recording_finished': {'stop_reason', 'stopped_by_signal'}, 'draining_started': set(), 'closed': set()}
        event = row.get('event')
        require(event in extra and set(row) == basic | extra[event], 'event_allowlist')
        if event == 'command_sent':
            require(row['command'] in ('0x81', '0x82', '0x83') and row['bytes_reported'] == 65 and (row['report_hex'] == reader.command_report(int(row['command'], 16)).hex()), 'command_proof')
        elif event == 'recording_finished':
            require(row['stop_reason'] in ('duration', 'stop_file') and row['stopped_by_signal'] is None, 'finish_event')
        elif event == 'health':
            require(set(row['since_last_health']) == {'voltage_V', 'current_A', 'power_W_calculated'}, 'health_metric_keys')
            for value in row['since_last_health'].values():
                stats(value)
            finite_tree({k: row[k] for k in HEALTH_KEYS - {'since_last_health'}})
        else:
            finite_tree({k: row[k] for k in extra[event]})

def gzip_exact(source, dest, kind, expected_sha, reader):
    """Preserve each original raw JSONL byte; deterministic gzip header."""
    h = hashlib.sha256()
    lines = 0
    with regular(source).open('rb') as src, dest.open('xb') as raw:
        with gzip.GzipFile(filename='', fileobj=raw, mode='wb', compresslevel=9, mtime=0) as gz:
            while True:
                line = src.readline(2000001)
                if not line:
                    break
                require(len(line) <= 2000000 and line.endswith(b'\n'), 'raw_JSONL_line_bound')
                row = strict_json(line)
                require(type(row) is dict, 'raw_object')
                validate_raw_row(kind, row, reader)
                h.update(line)
                gz.write(line)
                lines += 1
    require(h.hexdigest() == expected_sha, 'raw_changed_since_analysis_replay')
    return {'uncompressed_sha256': h.hexdigest(), 'lines': lines, 'gzip_mtime': 0, 'gzip_original_filename': '', 'raw_bytes_preserved': True}

def write(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write('\n')

def write_rows(path, rows):
    with path.open('x') as stream:
        for row in rows:
            stream.write(json.dumps(row, allow_nan=False) + '\n')

def source_text(path):
    """AST-only review copy; remove comments and private absolute host literals."""
    tree = ast.parse(regular(path, 2000000).read_text())

    class Public(ast.NodeTransformer):

        def visit_Constant(self, node):
            if isinstance(node.value, str) and ('<PRIVATE_HOST_PATH>' in node.value or '<PRIVATE_HOST_PATH>' in node.value):
                return ast.copy_location(ast.Constant('<PRIVATE_HOST_PATH>'), node)
            return node
    tree = Public().visit(tree)
    text = ast.unparse(tree) + '\n'
    require('<PRIVATE_HOST_PATH>' not in text and '<PRIVATE_HOST_PATH>' not in text, 'private_host_path_in_source')
    return '# Public AST review copy; comments/host path literals redacted. Original SHA is in export manifest.\n' + text
