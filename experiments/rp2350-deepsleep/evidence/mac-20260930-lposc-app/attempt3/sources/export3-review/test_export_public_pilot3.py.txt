class ExportTests(unittest.TestCase):

    @classmethod
    def setUpClass(cls):
        cls.tmp = tempfile.TemporaryDirectory(prefix='synthetic-export-', dir=ex.HERE)
        cls.root = Path(cls.tmp.name)
        cls.args = fixture(cls.root)

    @classmethod
    def tearDownClass(cls):
        cls.tmp.cleanup()

    def test_positive_closed_export_raw_bytes_and_no_identity(self):
        result = ex.export(self.args)
        self.assertEqual(result['result'], 'PASS')
        output = self.args.output
        for kind in ('frames', 'samples', 'events'):
            self.assertEqual(gzip.decompress((output / (kind + '.jsonl.gz')).read_bytes()), (self.root / 'pilot3/pilot-meter' / (kind + '.jsonl')).read_bytes())
        for path in output.rglob('*'):
            if path.is_file() and path.suffix != '.gz':
                text = path.read_text()
                forbidden_values = ['<PRIVATE_HOST_PATH>', '<PRIVATE_HOST_PATH>']
                if 'sources' not in path.relative_to(output).parts:
                    forbidden_values += [fixture_module.CANARY, fixture_module.RUN]
                for forbidden in forbidden_values:
                    self.assertNotIn(forbidden, text)
        self.assertFalse(any((p.name == 'app_image.py.txt' for p in output.rglob('*'))))
        self.assertIs(result['private_decoder_source_exported'], False)
        self.assertNotIn('hid', sys.modules)
        self.assertNotIn('serial', sys.modules)
        with self.assertRaises(ValueError):
            ex.export(self.args)

    def test_recovery_not_completed(self):
        path = self.args.restoration_result
        saved = path.read_bytes()
        value = json.loads(saved)
        value['purpose'] = 'recovery'
        dump(path, value)
        try:
            with self.assertRaises(ValueError):
                ex.closed_proofs(self.args)
        finally:
            path.write_bytes(saved)

    def test_worker_timeout_nonzero_or_unbound_analysis_rejected(self):
        path = self.args.root_completion
        saved = path.read_bytes()
        try:
            for field, value in [('timed_out', True), ('exit_code', 1), ('signal', 15), ('analysis_sha256', 'f' * 64)]:
                row = json.loads(saved)
                row[field] = value
                dump(path, row)
                with self.assertRaises(ValueError):
                    ex.closed_proofs(self.args)
        finally:
            path.write_bytes(saved)

    def test_missing_public_completion_before_private_read(self):
        path = self.args.root_completion
        saved = path.read_bytes()
        path.unlink()
        try:
            with self.assertRaises(ValueError):
                ex.closed_proofs(self.args)
        finally:
            path.write_bytes(saved)

    def test_private_backup_mismatch_rejected(self):
        path = self.args.restoration_result.parent / 'final-state-private.json'
        saved = path.read_bytes()
        row = json.loads(saved)
        row['state']['backup'][2][0] += 1
        dump(path, row)
        try:
            _, post, _, _, restoration = ex.closed_proofs(self.args)
            with self.assertRaises(ValueError):
                ex.restore_projection(self.args, post, restoration)
        finally:
            path.write_bytes(saved)

    def test_summary_text_in_numeric_field_rejected(self):
        summary = ex.read(self.root / 'pilot3/pilot-meter/summary.json')
        for field in ('requested_seconds', 'elapsed_host_s', 'empty_timed_reads', 'requested_duration_completed'):
            wrong = summary | {field: fixture_module.CANARY}
            with self.assertRaises(ValueError):
                ex.summary_projection(wrong, {'full_capture_quality': {}})

    def test_raw_unknown_identity_and_crc_rejected(self):
        reader, _ = analysis.load_frozen(ex.OLD)
        path = self.root / 'pilot3/pilot-meter/frames.jsonl'
        with path.open() as stream:
            first = json.loads(stream.readline())
        with self.assertRaises(ValueError):
            ex.validate_raw_row('frames', first | {'uid': fixture_module.CANARY}, reader)
        corrupt = first.copy()
        raw = bytearray.fromhex(first['raw_hex'])
        raw[12] ^= 1
        corrupt['raw_hex'] = raw.hex()
        with self.assertRaises(ValueError):
            ex.validate_raw_row('frames', corrupt, reader)

    def test_gzip_deterministic_header_and_allowlist(self):
        reader, _ = analysis.load_frozen(ex.OLD)
        source = self.root / 'pilot3/pilot-meter/events.jsonl'
        for name in ('copy1.gz', 'copy2.gz'):
            ex.gzip_exact(source, self.root / name, 'events', ex.sha(source), reader)
        self.assertEqual((self.root / 'copy1.gz').read_bytes(), (self.root / 'copy2.gz').read_bytes())
        self.assertEqual((self.root / 'copy1.gz').read_bytes()[4:8], b'\x00' * 4)

    def test_duplicate_json_key_cannot_hide_preserved_raw_text(self):
        reader, _ = analysis.load_frozen(ex.OLD)
        source = self.root / 'duplicate.jsonl'
        source.write_text('{"event":"PRIVATE_HIDDEN_VALUE","event":"closed","host_monotonic_ns":1000000000000,"host_receipt_utc":"2026-09-30T00:00:00+00:00"}\n')
        with self.assertRaises(ValueError):
            ex.gzip_exact(source, self.root / 'duplicate.gz', 'events', ex.sha(source), reader)
if __name__ == '__main__':
    unittest.main()
