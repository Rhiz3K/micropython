"""Install/observe the opt-in RP2350 reset-spanning test. Never flashes firmware."""

import argparse
import hashlib
import json
from pathlib import Path
import sys
import time
import os
import subprocess

HERE = Path(__file__).resolve().parent
PREFIX = b"DEEPSLEEP_TEST "
RADIO_MODES = ("never", "sta", "ap", "ble", "sta_ble", "sta_ap")


def validate_radio_config(config):
    mode = config.get("radio")
    if mode is None:
        if "ap" in config:
            raise ValueError("ap configuration requires radio=ap or sta_ap")
        return
    if mode not in RADIO_MODES:
        raise ValueError("unknown radio test mode")
    if "wifi" in config and mode not in ("sta", "sta_ble"):
        raise ValueError("wifi transaction requires radio=sta or sta_ble")
    if mode in ("ap", "sta_ap"):
        ap = config.get("ap")
        if not isinstance(ap, dict):
            raise ValueError("AP test requires private ap configuration")
        ssid = ap.get("ssid")
        if not isinstance(ssid, str) or not ssid.startswith("rp2350-sleep-test"):
            raise ValueError("AP SSID must start with rp2350-sleep-test")
        if not 1 <= len(ssid.encode()) <= 32:
            raise ValueError("AP SSID must fit 32 bytes")
        password = ap.get("password")
        if not isinstance(password, str) or not 8 <= len(password) <= 63:
            raise ValueError("AP password must contain 8..63 printable ASCII characters")
        if any(not 32 <= ord(char) <= 126 for char in password):
            raise ValueError("AP password must contain 8..63 printable ASCII characters")
    elif "ap" in config:
        raise ValueError("ap configuration requires radio=ap or sta_ap")


def validate_watchdog_config(config):
    after_reset = config.get("watchdog_after_reset", False)
    if type(after_reset) is not bool:
        raise ValueError("watchdog_after_reset must be boolean")
    if not after_reset:
        return
    if config.get("watchdog") is not True:
        raise ValueError("watchdog_after_reset requires watchdog=true")
    cycles = config.get("cycles", 100)
    if type(cycles) is not int or cycles < 1:
        raise ValueError("watchdog recovery requires a positive integer cycle count")
    if config.get("mode", "deepsleep") != "deepsleep":
        raise ValueError("watchdog recovery requires mode=deepsleep")
    if not config.get("expect_deep_cause", True) or config["sleep_ms"] < 2:
        raise ValueError("watchdog recovery requires timed sleep with DEEPSLEEP_RESET")


def expected_cycles(config):
    if config.get("watchdog") and not config.get("watchdog_after_reset"):
        return 0
    return config.get("cycles", 100)


def ports():
    from serial.tools import list_ports

    return list(list_ports.comports())


def validate_expected_record(record, config):
    if config is None:
        return
    if record.get("run") != config["run"]:
        raise ValueError("record does not belong to the expected run")
    if config.get("radio") is not None and record.get("radio_mode") != config["radio"]:
        raise ValueError("record radio mode does not match configuration")
    if config.get("watchdog_after_reset") and record.get("watchdog_after_reset") is not True:
        raise ValueError("record lacks watchdog-after-reset scenario marker")
    if record.get("status") == "PASS":
        expected = expected_cycles(config)
        if type(record.get("completed")) is not int or record["completed"] != expected:
            raise ValueError("PASS completed count does not match configuration")


def parse_record(line):
    index = line.find(PREFIX)
    if index < 0:
        return None
    record = json.loads(line[index + len(PREFIX) :])
    if not isinstance(record, dict) or record.get("status") not in (
        "READY",
        "SLEEP",
        "PASS",
        "FAIL",
        "STOPPED",
    ):
        raise ValueError("invalid test record/status")
    if record["status"] in ("FAIL", "STOPPED"):
        return record
    if (
        not isinstance(record.get("run"), str)
        or not record["run"]
        or type(record.get("boot")) is not int
        or record["boot"] < 1
        or type(record.get("completed")) is not int
        or record["completed"] < 0
        or record.get("protocol") not in (None, 2)
    ):
        raise ValueError("invalid test record fields")
    if record["status"] != "SLEEP" or record.get("protocol") == 2:
        if not isinstance(record.get("uid"), str):
            raise ValueError("missing target identity")
    return record


def acknowledgement(record):
    if record.get("protocol") == 2:
        return "GO {} {} {}\n".format(record["run"], record["boot"], record["status"])
    return "GO " + record["run"] + "\n"


class RunProgress:
    """Validate the full observed sequence, not just the final device counter."""

    def __init__(self, config):
        self.config = config
        self.ready = None
        self.sleep_started = None
        self.watchdog_reset_seen = False

    def check(self, record, now):
        if self.config is None:
            return
        if record.get("protocol") != 2:
            raise ValueError("configured run requires protocol 2; reinstall current device.py")
        current = (record["boot"], record["completed"])
        status = record["status"]
        watchdog = self.config.get("watchdog", False)
        after_reset = self.config.get("watchdog_after_reset", False)
        if status == "READY":
            if self.ready is None:
                if current[1] != 0 or record.get("phase") != 0:
                    raise ValueError("missing initial READY; cannot verify a partial run")
            elif current == self.ready:
                if self.sleep_started is not None:
                    raise ValueError("READY after SLEEP on the same boot")
                return
            elif after_reset and not self.watchdog_reset_seen and record.get("phase") == 4:
                if (
                    self.sleep_started is not None
                    or self.ready[1] != 0
                    or current != (self.ready[0] + 1, 0)
                    or type(record.get("watchdog_reason")) is not int
                    or not record["watchdog_reason"] & 1
                    or type(record.get("watchdog_ctrl")) is not int
                    or record["watchdog_ctrl"] & (1 << 30)
                ):
                    raise ValueError("watchdog timeout does not follow the initial READY")
                self.watchdog_reset_seen = True
            else:
                if (
                    (watchdog and not after_reset)
                    or (after_reset and not self.watchdog_reset_seen)
                    or self.sleep_started is None
                    or current != (self.ready[0] + 1, self.ready[1] + 1)
                    or record.get("phase") != 1
                ):
                    raise ValueError("wake does not match the acknowledged sleep")
                elapsed = now - self.sleep_started
                target = self.config["sleep_ms"] / 1000
                tolerance = self.config.get("host_tolerance_s", max(0.25, target * 0.1))
                record["host_cycle_elapsed_s"] = elapsed
                if elapsed < max(0, target - tolerance):
                    raise ValueError("wake arrived before the host duration lower bound")
                self.sleep_started = None
            self.ready = current
        elif status == "SLEEP":
            if (
                (watchdog and not after_reset)
                or (after_reset and not self.watchdog_reset_seen)
                or current != self.ready
                or current[1] >= self.config.get("cycles", 100)
                or record.get("sleep_ms") != self.config["sleep_ms"]
            ):
                raise ValueError("SLEEP does not match READY/configuration")
        elif status == "PASS":
            expected = expected_cycles(self.config)
            if (
                self.ready is None
                or (after_reset and not self.watchdog_reset_seen)
                or self.sleep_started is not None
                or self.ready[1] != expected
                or current != (self.ready[0] + 1, expected)
                or record.get("phase") != (4 if watchdog and not after_reset else 5)
            ):
                raise ValueError("PASS lacks a complete observed run and final reset")

    def acknowledged(self, record, now):
        if self.config is not None and record["status"] == "SLEEP" and self.sleep_started is None:
            self.sleep_started = now


def log_fragment(log, buffer, started):
    if PREFIX in buffer:
        log.write(
            json.dumps(
                {
                    "host_event": "truncated_record_on_disconnect",
                    "line_bytes": len(buffer),
                    "host_elapsed_s": time.monotonic() - started,
                }
            )
            + "\n"
        )


def load_manifest(path):
    data = json.loads(path.read_text())
    if not data.get("authorized") or not data.get("bootsel_recovery_confirmed"):
        raise ValueError("manifest must attest authorization and BOOTSEL recovery")
    for key in ("usb_serial", "unique_id", "machine_contains"):
        if not data.get(key):
            raise ValueError("missing identity field: " + key)
    for key in ("firmware_backup", "filesystem_backup"):
        backup = data[key]
        source = (path.parent / backup["path"]).resolve()
        digest = hashlib.sha256(source.read_bytes()).hexdigest()
        if digest != backup["sha256"]:
            raise ValueError("backup checksum mismatch: " + str(source))
    return data


def port_for(manifest):
    found = [p for p in ports() if p.serial_number == manifest["usb_serial"]]
    if len(found) > 1:
        raise ValueError("USB serial is not unique")
    return found[0].device if found else None


def timing_ack(connection, record, log, ack_times):
    """One bounded write per qualified marker; never reset or silently retry it."""
    key = (record["run"], record["boot"], record["status"])
    if key in ack_times:
        return ack_times[key]
    payload = acknowledgement(record).encode()
    before_ns = time.monotonic_ns()
    wall_ns = time.time_ns()
    try:
        count = connection.write(payload)
    except Exception:
        # Serial errors after a possible partial ACK cannot be re-timed safely.
        raise RuntimeError("Timing ACK write failed; run is incomplete") from None
    after_ns = time.monotonic_ns()
    if count != len(payload):
        raise RuntimeError("Partial timing ACK write; run is incomplete")
    ack_times[key] = before_ns
    log.write(json.dumps({"host_event": "timing_ack", "run": key[0], "boot": key[1],
        "marker_status": key[2], "host_ack_before_ns": before_ns,
        "host_ack_after_ns": after_ns, "host_ack_wall_ns": wall_ns,
        "write_bytes": count, "write_span_ns": after_ns-before_ns}) + "\n")
    return before_ns


def observe(args, manifest):
    import serial

    if not args.allow_run:
        raise ValueError("sleep/reset execution requires --allow-run")
    expected = None
    if args.config is not None:
        expected = json.loads(args.config.read_text())
        if (
            not expected.get("run")
            or type(expected.get("cycles", 100)) is not int
            or expected.get("cycles", 100) < 1
        ):
            raise ValueError("expected config requires a run name and positive cycle count")
        validate_radio_config(expected)
        if (
            type(expected.get("sleep_ms")) is not int
            or not 0 <= expected["sleep_ms"] <= 2147483647
        ):
            raise ValueError("expected config requires sleep_ms in 0..2147483647")
        validate_watchdog_config(expected)
        tolerance = expected.get("host_tolerance_s", 0)
        if not isinstance(tolerance, (int, float)) or not 0 <= tolerance < float("inf"):
            raise ValueError("host_tolerance_s must be finite and nonnegative")
    progress = RunProgress(expected)
    seen = set()
    ack_times = {}
    started = time.monotonic()
    last_progress = started
    connection = None
    buffer = b""
    with args.log.open("x", buffering=1) as log:
        log.write(
            json.dumps({"host_start": time.time(), "usb_serial": manifest["usb_serial"]}) + "\n"
        )
        try:
            while time.monotonic() - last_progress < args.timeout:
                try:
                    if connection is None:
                        port = port_for(manifest)
                        if port is None:
                            time.sleep(0.2)
                            continue
                        connection = serial.Serial(port, 115200, timeout=0.05, write_timeout=1.0, exclusive=True)
                        buffer = b""
                        # Establish host presence after DTR/enumeration. This
                        # also terminates a partial line from an earlier client.
                        connection.write(b"\nHELLO\n")  # bounded write; no tcdrain/flush
                    buffer += connection.read(connection.in_waiting or 1)
                    read_ns = time.monotonic_ns()
                    read_wall_ns = time.time_ns()
                    while b"\n" in buffer:
                        line, buffer = buffer.split(b"\n", 1)
                        index = line.find(PREFIX)
                        if index < 0:
                            continue
                        try:
                            record = parse_record(line)
                        except (ValueError, UnicodeError):
                            # A disconnect may truncate JSON or split a UTF-8
                            # character. Never acknowledge it or extend timeout.
                            log.write(
                                json.dumps(
                                    {
                                        "host_event": "malformed_record",
                                        "line_bytes": len(line),
                                        "host_elapsed_s": time.monotonic() - started,
                                    }
                                )
                                + "\n"
                            )
                            continue
                        record["host_elapsed_s"] = time.monotonic() - started
                        record["host_read_monotonic_ns"] = read_ns
                        record["host_read_wall_ns"] = read_wall_ns
                        try:
                            validate_expected_record(record, expected)
                        except ValueError:
                            log.write(json.dumps(record) + "\n")
                            raise
                        status = record.get("status")
                        if status in ("FAIL", "STOPPED"):
                            log.write(json.dumps(record) + "\n")
                            raise RuntimeError("Device reported FAIL/STOPPED; private log preserves evidence")
                        if (status != "SLEEP" or record.get("protocol") == 2) and (
                            record.get("uid", "").lower() != manifest["unique_id"].lower()
                        ):
                            raise ValueError("reconnected target identity mismatch")
                        try:
                            progress.check(record, time.monotonic() - started)
                        except ValueError:
                            log.write(json.dumps(record) + "\n")
                            raise
                        if status == "SLEEP":
                            log.write(json.dumps(record) + "\n")
                            if record.get("protocol") == 2:
                                ack_ns = timing_ack(connection, record, log, ack_times)
                                progress.acknowledged(record, ack_ns / 1e9 - started)
                            continue
                        key = (record["run"], record["boot"])
                        if key not in seen:
                            seen.add(key)
                            last_progress = time.monotonic()
                            log.write(json.dumps(record) + "\n")
                            print(json.dumps({k:record[k] for k in ("run","boot","completed","status","cause","host_elapsed_s") if k in record}), flush=True)
                        timing_ack(connection, record, log, ack_times)
                        if status == "PASS":
                            return
                    if len(buffer) > 65536:
                        raise RuntimeError("unbounded non-test serial output")
                    if port_for(manifest) is None:
                        log_fragment(log, buffer, started)
                        connection.close()
                        connection = None
                        buffer = b""
                except (OSError, serial.SerialException):
                    log_fragment(log, buffer, started)
                    buffer = b""
                    if connection is not None:
                        connection.close()
                        connection = None
                    time.sleep(0.2)
            raise TimeoutError("No boot progress within timeout; log is incomplete, not PASS")
        finally:
            if connection is not None:
                connection.close()


FW_SHA = 'a0a2107db05f6ea2246cb8540e2d715b72bbcfa7a0cae1492b25cb6b7ae2ee08'
FW_SIZE = 889840
NEW_VERSION = 'v1.30.0-preview.105.g6ce738ae0f.dirty'
_BaseRunProgress = RunProgress
class TimingProgress(_BaseRunProgress):
    def check(self, record, now):
        status = record.get("status")
        if status in ("READY", "PASS"):
            if (record.get("fixture_bin_size") != FW_SIZE
                    or record.get("fixture_bin_sha256") != FW_SHA
                    or record.get("fixture_runtime", "").split(" on ")[0] != NEW_VERSION):
                raise ValueError("Exact production BIN/runtime binding failed")
            panel = record.get("panel_prep", {})
            if not (panel.get("status") == "PREPARED" and panel.get("refreshed") is False
                    and panel.get("park_registers_verified") is True
                    and panel.get("vendor_sleep_completed") is True):
                raise ValueError("Panel prefix was not verified on this boot")
            if record.get("watchdog_ctrl", 1 << 30) & (1 << 30):
                raise ValueError("Watchdog is active")
        if status == "READY" and self.ready is None:
            if (record.get("boot"), record.get("phase"), record.get("completed")) != (1, 0, 0):
                raise ValueError("Exact first boot/initial READY is required")
        if status == "READY" and record.get("phase") == 1:
            if (record.get("cause") != 4
                    or not record.get("powman_chip_reset", 0) & (1 << 25)
                    or record.get("powman_last_swcore_pwrup") != 64):
                raise ValueError("Deep-wake cause/POWMAN flags mismatch")
        if status == "PASS" and record.get("cause") != 3:
            raise ValueError("Final ordinary reset cause mismatch")
        return super().check(record, now)

RunProgress = TimingProgress

def main():
    os.umask(0o077)
    parser = argparse.ArgumentParser(description="Bounded private protocol-2 timing observer; never installs or resets")
    parser.add_argument("--manifest", type=Path, required=True)
    parser.add_argument("--seconds", type=int, choices=(30,300), required=True)
    parser.add_argument("--allow-run", action="store_true")
    parser.add_argument("--worker", action="store_true", help=argparse.SUPPRESS)
    args = parser.parse_args()
    args.manifest = args.manifest.resolve()
    args.config = HERE / ("config-%ds.json" % args.seconds)
    args.log = HERE / ("%ds-events-private.jsonl" % args.seconds)
    args.timeout = args.seconds * 1.10 + 60
    if not args.allow_run:
        raise ValueError("Explicit --allow-run is required")
    if args.log.exists():
        raise FileExistsError("Timing log already exists; refusing overwrite or resume")
    if not args.worker:
        command = [sys.executable, str(Path(__file__).resolve()), "--worker", "--manifest", str(args.manifest),
                   "--seconds", str(args.seconds), "--allow-run"]
        try:
            return subprocess.run(command, timeout=2*args.seconds*1.10+180).returncode
        except subprocess.TimeoutExpired:
            print(json.dumps({"result":"INCOMPLETE", "reason":"outer_worker_timeout", "seconds":args.seconds}), flush=True)
            return 124  # No device reset, cleanup, retry, or reconnect is performed here.
    config = json.loads(args.config.read_text())
    if config.get("cycles") != 2 or config.get("sleep_ms") != args.seconds*1000:
        raise ValueError("Expected exactly two cycles of the selected duration")
    if config.get("radio") != "never" or config.get("mode", "deepsleep") != "deepsleep":
        raise ValueError("Timing requires deepsleep and radio=never")
    if any(k in config for k in ("wifi", "watchdog", "watchdog_after_reset", "thread", "dma", "lightsleep_before")):
        raise ValueError("Other test scenarios are forbidden")
    observe(args, load_manifest(args.manifest))
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
