 p.require(proof['finished_monotonic_ns']-proof['boundaries'][-1]['receipt_monotonic_ns']>=12_000_000_000,'Application listener tail incomplete')
 installed=p.mapping(p.read_json(ROOT/'pilot-installed-inventory-private.json'))
 b,ident=p.connect(p.NEW_VERSION)
 try:
  rows=p.inventory(b);files=p.mapping(rows)
  expected=dict(installed);expected[CONSUMED]=expected.pop(ARMED)
  extras=set(files)-set(expected)
  p.require(extras=={EVENTS,TERMINAL} and all(files.get(k)==v for k,v in expected.items()),'Pilot original/source file drift')
  p.require(files[EVENTS][0]<=32768 and files[TERMINAL][0]<=2048 and p.directories(b)==p.read_json(ROOT/'directories-before-private.json'),'Pilot output/directory bounds')

   b.fs_get(device,str(ROOT/local));p.require((ROOT/local).stat().st_size==files[device][0] and p.sha(ROOT/local)==files[device][1],'Pilot mutable output readback')
  terminal=p.read_json(ROOT/'pilot-device-terminal-private.json')

def run():
 p.require((ROOT/'pilot-preflight-private.json').is_file() and not (ROOT/'pilot-go-proof-private.json').exists(),'One-shot pilot preflight')
 import importlib.util
 spec=importlib.util.spec_from_file_location('pilot_listener',REVIEW/'marker_listener.py');listenerlib=importlib.util.module_from_spec(spec);spec.loader.exec_module(listenerlib)
 cfg=p.read_json(ROOT/'pilot-control-private.json')
 capture=ROOT/'pilot-meter';markerout=ROOT/'pilot-markers';stop=ROOT/'pilot-meter.stop'
 reader=None;listener=None;caffeinate=None
 files=[]
 try:
  with contextlib.ExitStack() as stack:
   for name in ('.measure-variants.lock','.wake-rtc.lock'):
    lock=stack.enter_context((p.OLD/name).open('a+'));fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
   caffeinate=subprocess.Popen(['/usr/bin/caffeinate','-dims','-w',str(os.getpid())],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
   meterlog=stack.enter_context((ROOT/'pilot-meter-process-private.log').open('xb'))
   reader=subprocess.Popen([sys.executable,str(READER),'--seconds','1800','--identity-json',str(ROOT/'meter-identity-private.json'),'--output-dir',str(capture),'--stop-file',str(stop)],stdout=meterlog,stderr=subprocess.STDOUT)
   deadline=time.monotonic()+25
   while listenerlib.meter_ready(capture,time.monotonic_ns()) is None:
    p.require(reader.poll() is None and time.monotonic()<deadline,'Meter startup failed; no Pico launch')
    time.sleep(.1)
   b,ident=p.connect(p.NEW_VERSION)
   try:
    p.require(p.mapping(p.inventory(b))==p.mapping(p.read_json(ROOT/'pilot-installed-inventory-private.json')) and p.firmware_hash(b,p.FW_SIZE)==p.FW_SHA and not p.state(b)['wdt_ctrl']&(1<<30),'Fresh GO identity/BIN/files gate')
    stamp=time.monotonic_ns()
    proof=dict(schema=2,run=cfg['run'],approved=True,usb_mode='USB_CONNECTED',usb_connected_verified=True,pico_identity_verified=True,firmware_verified=True,installed_files_verified=True,firmware_version=p.NEW_VERSION,firmware_bin_sha256=p.FW_SHA,root_preflight_sha256=p.sha(ROOT/'pilot-preflight-private.json'),board_verified_monotonic_ns=stamp)
    p.write_json(ROOT/'pilot-go-proof-private.json',proof)
    listenerlog=stack.enter_context((ROOT/'pilot-listener-process-private.log').open('xb'))
    listener=subprocess.Popen([sys.executable,str(REVIEW/'marker_listener.py'),'--bind-ip',cfg['marker_ipv4'],'--port',str(cfg['marker_port']),'--run',cfg['run'],'--output',str(markerout),'--go-proof',str(ROOT/'pilot-go-proof-private.json'),'--preflight-proof',str(ROOT/'pilot-preflight-private.json'),'--firmware-version',p.NEW_VERSION,'--firmware-sha256',p.FW_SHA,'--meter-dir',str(capture)],stdout=listenerlog,stderr=subprocess.STDOUT)
    wait=time.monotonic()+5
    while not (markerout/'events.jsonl').exists():
     p.require(listener.poll() is None and time.monotonic()<wait,'LAN listener startup failed')
     time.sleep(.05)
    b.exec_raw_no_follow('time.sleep_ms(150); machine.reset()')
   finally:b.close()
   print(json.dumps(dict(result='PILOT_LAUNCHED',usb_mode='USB_CONNECTED',cycles=3,sleep_ms=300000)),flush=True)
   first_limit=time.monotonic()+120;last_progress=time.monotonic();count=0
   while listener.poll() is None:
    p.require(reader.poll() is None,'Meter reader failed during application')
    path=markerout/'events.jsonl'
    rows=[]
    if path.exists():
     for line in path.read_text().splitlines():
      try:rows.append(json.loads(line))
      except json.JSONDecodeError:continue
    boundaries=[r for r in rows if r.get('kind')=='DOWNLOAD_READY' and r.get('duplicate') is False]
    if len(boundaries)>count:
     count=len(boundaries);last_progress=time.monotonic()
     print(json.dumps(dict(event='DOWNLOAD_BOUNDARY',index=count-1,usb_mode='USB_CONNECTED')),flush=True)
    p.require(time.monotonic()<(first_limit if count==0 else last_progress+480),'Application marker deadline; incomplete no retry')
    time.sleep(.2)
   complete=listener.returncode==0 and p.read_json(markerout/'result.json')['result']=='BOUNDARIES_COMPLETE'
   stop.write_bytes(b'root bounded application pilot tail complete\n')
   reader.wait(timeout=15)
   p.write_json(ROOT/'pilot-process-public.json',dict(result='PASS' if complete and reader.returncode==0 else 'FAIL',listener_boundaries_complete=complete,capture_exit=reader.returncode,usb_mode='USB_CONNECTED',**p.stamp()))
   p.require(complete,'Application marker listener failed; no automatic REPL or device retry')
   postcheck(True)
 finally:
  if listener is not None and listener.poll() is None:
   listener.terminate()
   try:listener.wait(timeout=5)
   except subprocess.TimeoutExpired:listener.kill();listener.wait()
  if reader is not None and reader.poll() is None:
   if not stop.exists():stop.write_bytes(b'root incomplete pilot stop\n')
   try:reader.wait(timeout=15)
   except subprocess.TimeoutExpired:reader.terminate();reader.wait(timeout=5)
  if caffeinate is not None and caffeinate.poll() is None:caffeinate.terminate()

def main():
 p.require(sys.argv[1] in ('stage','run'),'Explicit pilot phase')
 phase=sys.argv[1]
 if len(sys.argv)==3 and sys.argv[2]=='--worker':
  globals()[phase]()
  return
 p.require(len(sys.argv)==2,'Invalid arguments')
 child=subprocess.Popen([sys.executable,str(Path(__file__).resolve()),phase,'--worker'],start_new_session=True)
 try:code=child.wait(timeout=1950 if phase=='run' else 120)
 except subprocess.TimeoutExpired:
  os.killpg(child.pid,signal.SIGTERM)
  try:child.wait(timeout=5)
  except subprocess.TimeoutExpired:os.killpg(child.pid,signal.SIGKILL);child.wait()
  raise RuntimeError('Pilot root phase deadline; no automatic device retry') from None
 raise SystemExit(code)

if __name__=='__main__':main()
