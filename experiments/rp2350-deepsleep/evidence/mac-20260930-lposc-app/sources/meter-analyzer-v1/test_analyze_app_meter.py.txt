def utc(seconds):
    return (EPOCH + dt.timedelta(seconds=seconds)).isoformat()


def dump(path, value):
    path.write_text(json.dumps(value) + '\n')


def rows(path, values):
    path.write_text(''.join(json.dumps(x) + '\n' for x in values))

class OfflineTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.tmp = tempfile.TemporaryDirectory(prefix='synthetic-', dir=HERE)
        cls.args, cls.meta, cls.result, cls.markers, cls.device, cls.terminal, cls.events = fixture(Path(cls.tmp.name))
        cls.m, cls.a = app.load_frozen(OLD)

    @classmethod
    def tearDownClass(cls):
        cls.tmp.cleanup()

    def marker_check(self, rows=None, result=None, meta=None):
        return app.validate_marker(meta or self.meta, result or self.result, rows or self.markers,
            app.sha(self.args.meter_dir / 'metadata.json'), FW, app.sha(self.args.preflight_proof))

    def test_end_to_end_retains_zero_spike_and_sanitizes(self):
        result = app.analyze(self.args)
        self.assertEqual(result['result'], 'PASS')
        self.assertEqual(len(result['periods']), 3)
        first = result['periods'][0]
        self.assertEqual(first['samples'], 30000)
        self.assertEqual(first['samples_exactly_zero_current'], 1)
        self.assertEqual(first['current_A']['min'], 0)
        self.assertEqual(first['current_A']['max'], .5)
        energy = first['integral_estimates']
        self.assertAlmostEqual(energy['host_receipt_frame_mean_trapezoid']['energy_J'], 1.5249, places=9)
        self.assertAlmostEqual(energy['nominal_10ms_per_in_window_received_sample']['energy_J'], 1.5249, places=9)
        self.assertEqual(result['full_capture_quality']['samples_exactly_zero_current'], 1)
        self.assertFalse(result['valid_sleep_only_energy'])
        text = (self.args.output / 'analysis-public.json').read_text()
        for secret in (CANARY, RUN, str(HERE), 'expected_serial_number'):
            self.assertNotIn(secret, text)
        self.assertFalse(any(x in __import__('sys').modules for x in ('hid', 'serial')))

    def test_duplicate_keeps_first_boundary(self):
        values = copy.deepcopy(self.markers)
        dup = copy.deepcopy(values[1])
        for key in ('receipt_monotonic_ns', 'ack_before_monotonic_ns', 'ack_after_monotonic_ns'):
            dup[key] += 5_000_000
        dup['duplicate'] = True
        values.insert(2, dup)
        _, boundaries, stats = self.marker_check(rows=values)
        self.assertEqual(boundaries[0], self.markers[1])
        self.assertEqual(stats['duplicate_datagrams'], 1)

    def test_missing_boundary_rejected(self):
        with self.assertRaises(ValueError): self.marker_check(rows=self.markers[:-1])

    def test_stale_live_gate_rejected(self):
        values = copy.deepcopy(self.markers)
        values[2]['meter_gate']['health_monotonic_ns'] -= 4 * app.NS
        with self.assertRaises(ValueError): self.marker_check(rows=values)

    def test_listener_fail_rejected(self):
        result = copy.deepcopy(self.result); result['result'] = 'FAIL'
        with self.assertRaises(ValueError): self.marker_check(result=result)

    def test_device_final_and_cause_required(self):
        for value in (self.device[:-1], self.device[:3] + self.device[4:]):
            with self.assertRaises(ValueError): app.validate_device(value, self.terminal, RUN)
        wrong = copy.deepcopy(self.device); wrong[3]['cause'] = 3
        with self.assertRaises(ValueError): app.validate_device(wrong, self.terminal, RUN)
        with self.assertRaises(ValueError): app.validate_device(self.device, {'schema': 1, 'result': 'FAIL', 'index': 3}, RUN)

    def test_presence_flags_are_observations(self):
        proof = app.validate_device(self.device, self.terminal, RUN)
        self.assertIs(proof['downloads'][0]['server_partial_present'], False)

    def test_capture_error_clock_gap_events_rejected(self):
        summary = app.read_json(self.args.meter_dir / 'summary.json')
        for kind in ('error', 'drain_error', 'invalid_frame', 'receipt_gap', 'wall_vs_monotonic_clock_offset_jump'):
            with self.assertRaises(ValueError): app.check_meter_events(self.a, self.events + [{'event': kind}], summary)

    def test_health_cadence_rejected(self):
        values = copy.deepcopy(self.events)
        next(x for x in values if x['event'] == 'health')['recent_receipt_sample_rate_hz'] = 106
        with self.assertRaises(ValueError): app.check_meter_events(self.a, values, app.read_json(self.args.meter_dir / 'summary.json'))

    def test_host_clock_jump_and_different_boot_rejected(self):
        proof = {'meter_UTC_minus_monotonic_offset_change_s': {'min': 0, 'max': 0}}
        first = {'host_receipt_utc': utc(0), 'host_receipt_monotonic_ns': BASE}
        self.assertEqual(app.check_clock(self.a, proof, first, self.meta, self.result)['kind'], 'SAME_HOST_MONOTONIC_RECEIPT')
        changed = copy.deepcopy(proof); changed['meter_UTC_minus_monotonic_offset_change_s']['max'] = .101
        with self.assertRaises(ValueError): app.check_clock(self.a, changed, first, self.meta, self.result)
        first_wrong = first | {'host_receipt_monotonic_ns': BASE + app.NS}
        with self.assertRaises(ValueError): app.check_clock(self.a, proof, first_wrong, self.meta, self.result)
        wrong_end = self.result | {'finished_utc': utc(919)}
        with self.assertRaises(ValueError): app.check_clock(self.a, proof, first, self.meta, wrong_end)

    def test_capture_summary_failure_rejected(self):
        path = self.args.meter_dir / 'summary.json'
        original = path.read_bytes()
        for key, value in (('capture_result', 'ERROR'), ('clock_offset_jump_count', 1), ('data_quality', 'DEGRADED')):
            data = json.loads(original); data[key] = value; dump(path, data)
            try:
                with self.assertRaises(ValueError): app.analyze(self.args)
            finally: path.write_bytes(original)

    def test_frozen_window_rejects_gap_and_missing_edges(self):
        for connected, end in ((False, 1 * app.NS), (True, int(1.01 * app.NS))):
            window = self.a.Window('synthetic', 0, end)
            for n in range(26):
                t = n * 40_000_000
                values = [{'current_A': .001, 'voltage_V': 5., 'power_W_calculated': .005,
                           'current_raw': 100, 'voltage_raw': 500000}] * 4
                window.frame(t, values)
                if n: window.segment(((n - 1) * 40_000_000, .001, .005), (t, .001, .005), connected)
            self.assertEqual(window.result()['result'], 'FAIL')

    def test_raw_crc_and_exact_sample_proof_rejects_tamper(self):
        # One real synthetic frame/sample group passed to the same frozen proof.
        frames = iter(self.a.JsonLines(self.args.meter_dir / 'frames.jsonl'))
        first = next(frames); frames.close()
        samples = iter(self.a.JsonLines(self.args.meter_dir / 'samples.jsonl'))
        values = [next(samples) for _ in range(4)]; samples.close()
        ledger = self.m.StreamAccounting()
        parsed = self.m.parse_frame(bytes.fromhex(first['raw_hex']))
        ledger.add(parsed, 'recording', first['host_receipt_monotonic_ns'], 0)
        class NoBins:
            def add(self, *args): pass
        forged = copy.deepcopy(values); forged[0]['current_A'] = 999
        with self.assertRaises(ValueError): self.a.prove_samples([first], forged, [], NoBins(), ledger.snapshot())
        wrong_crc = copy.deepcopy(first); raw = bytearray.fromhex(first['raw_hex']); raw[12] ^= 1; wrong_crc['raw_hex'] = raw.hex()
        with self.assertRaises(ValueError): self.a.prove_samples([wrong_crc], values, [], NoBins(), ledger.snapshot())
