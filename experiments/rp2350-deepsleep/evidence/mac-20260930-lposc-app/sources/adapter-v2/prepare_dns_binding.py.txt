"""Offline AST-only private binding builder. No imports of original app code.

The exclusive output contains a private fallback address; never publish it.
Only its hash and safe source hashes belong in public evidence.
"""
import argparse
import ast
import hashlib
import ipaddress
import json
import os
from pathlib import Path
from urllib.parse import urlsplit


def build(original):
    original = Path(original)
    main = (original / 'main.py').read_bytes()
    config = (original / 'config.txt').read_bytes()
    cfg = {}
    for line in config.decode().splitlines():
        line = line.strip()
        if line and not line.startswith('#') and '=' in line:
            key, value = line.split('=', 1)
            cfg[key.strip()] = value.strip()
    url = urlsplit(cfg['ENDPOINT'])
    if url.scheme != 'http' or url.username or url.password or not url.hostname:
        raise ValueError('Endpoint structural gate')
    dictionaries = [ast.literal_eval(node.value) for node in ast.parse(main).body
                    if isinstance(node, ast.Assign) and any(
                        isinstance(t, ast.Name) and t.id == '_DNS_IP_FALLBACKS'
                        for t in node.targets)]
    if len(dictionaries) != 1 or not isinstance(dictionaries[0], dict):
        raise ValueError('Fallback AST gate')
    fallback = dictionaries[0].get(url.hostname)
    if fallback is not None and str(ipaddress.IPv4Address(fallback)) != fallback:
        raise ValueError('Fallback address gate')
    servers = [s.strip() for s in cfg.get('DNS_SERVERS', '').split(',') if s.strip()]
    if not 1 <= len(servers) <= 3 or any(str(ipaddress.IPv4Address(s)) != s for s in servers):
        raise ValueError('Configured DNS gate')
    return {'schema': 1, 'endpoint_host_sha256': hashlib.sha256(url.hostname.encode()).hexdigest(),
            'fallback_ipv4': fallback, 'main_source_sha256': hashlib.sha256(main).hexdigest(),
            'config_source_sha256': hashlib.sha256(config).hexdigest()}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--original', required=True, type=Path)
    parser.add_argument('--output', required=True, type=Path)
    args = parser.parse_args()
    value = build(args.original)
    fd = os.open(args.output, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, 'w') as stream:
        json.dump(value, stream, sort_keys=True)
        stream.write('\n')
    print('PRIVATE_BINDING_CREATED')


if __name__ == '__main__':
    main()
