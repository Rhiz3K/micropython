#!/usr/bin/env python3
"""Run five remaining profiles once; retain A PASS and two B-LS failures."""
import argparse
import fcntl
import json
import os
from pathlib import Path
import sys
import continue_comparisons2 as previous

c, base, ROOT = previous.c, previous.base, previous.ROOT
PREFIX = 'cmp300deep'
PRIOR = 'measurements/cmp300cont2'
B = PRIOR + '/02-b-lightsleep-reset'
RUN = 'runs/cmp300cont2-b-lightsleep-reset'
FREEZE = 'measurement-source-freeze-cont2.json'
FREEZE_SHA = '8b551a3a04de74288aa4fea5a9ac8b66f4dc05e1ee6c99bce9824c9d2c9745d4'
REVIEW = 'comparison-deep-continuation-review.json'
REVIEW_SHA = '303ad97f33cf63252fcb94be35a67f644588b0eced5e36eacacec69fd9efffe3'
PROOFS = (FREEZE, REVIEW, PRIOR + '/result.json', PRIOR + '/progress.json',
          PRIOR + '/supervision-events.jsonl', B + '/result.json',
          B + '/measurement-metadata.json', B + '/meter/summary.json',
          B + '/meter/health.json', RUN + '/result-public.json',
          RUN + '/config.json', RUN + '/plan.json')


def plan(root, prepared=None):
    original = base.plan(root, PREFIX)
    base.require(prepared is None or prepared == original, 'Unexpected base plan')
    return {**original, 'profiles': original['profiles'][2:], 'continuation': {
        'retained_A_directory': c.A, 'excluded_B_result': 'FAIL',
        'excluded_B_suite_failures': [c.B, B], 'B_valid_current': False, 'B_valid_energy': False,
        'previous_meter_startup_failure': previous.B,
        'root_review': REVIEW, 'root_review_sha256': REVIEW_SHA, 'profiles': 5,
        'canonical_directory_indices': [3, 4, 5, 6, 7],
        'note': 'B-LS failed in two actual suites and will not be rerun. A stays retained PASS. '
                'Only five other unchanged profiles; original V2 gates stop on any new failure.'}}


def verify_B_failure(root):
    base.require(base.sha(root / FREEZE) == FREEZE_SHA, 'Cont2 freeze changed')
    source = base.read_json(root / FREEZE)['source_sha256']
    base.require(len(source) == 45, 'Unexpected cont2 source/proof snapshot')
    base.verify_sources(root, source)
    base.require(base.sha(root / REVIEW) == REVIEW_SHA, 'Explicit deep continuation review changed')
    review = base.read_json(root / REVIEW)
    base.require(review.get('explicit_root_review_completed') is True
                 and review.get('prior_sequence') == 'cmp300cont2'
                 and review.get('B_LS_actual_attempts') == 2 and review.get('B_LS_result') == 'FAIL'
                 and review.get('B_LS_valid_current_or_energy') is False
                 and review.get('firmware_or_sleep_harness_changed') is False,
                 'Explicit deep continuation review missing')
    result = base.read_json(root / PRIOR / 'result.json')
    progress = base.read_json(root / PRIOR / 'progress.json')
    profile = base.read_json(root / B / 'result.json')
    base.require(result.get('result') == 'FAIL' and result.get('failed_phase') == 'suite'
                 and result.get('cleanup_errors') == [] and result.get('profiles') == [profile]
                 and progress.get('result') == 'FAIL' and progress.get('phase') == 'finished'
                 and progress.get('completed_profiles') == [], 'Cont2 is not cleanly closed suite FAIL')
    base.require(profile.get('result') == 'FAIL' and profile.get('failed_phase') == 'suite'
                 and profile.get('functional_result') == 'FAIL' and profile.get('capture_result') == 'PASS'
                 and profile.get('energy_result') == 'NOT_RUN' and profile.get('cleanup_errors') == []
                 and profile.get('run_label') == 'cmp300cont2-b-lightsleep-reset'
                 and profile.get('complete_cycle_energy_model_valid') is False,
                 'Cont2 B failure or invalid-energy status differs')
    suite = base.read_json(root / RUN / 'result-public.json')
    observed = suite.get('observed', {})
    base.require(suite.get('result') == 'FAIL' and observed.get('status') == 'FAIL'
                 and observed.get('boot') == 2 and observed.get('completed') == 0
                 and observed.get('rtc_elapsed_s') == 2, 'Second B suite failure differs')
    cfg = base.read_json(root / RUN / 'config.json')
    base.require(cfg == {'run': 'meter-20260929-cmp300cont2-b-lightsleep-reset',
        'cycles': 3, 'sleep_ms': 300000, 'radio': 'never', 'mode': 'lightsleep_reset',
        'expect_deep_cause': False, 'rtc_tolerance_s': 10, 'host_tolerance_s': 3},
        'Second B suite was not the unchanged profile')
    with (root / PRIOR / 'supervision-events.jsonl').open() as stream:
        launches = [e.get('name') for line in stream if (e := json.loads(line)).get('event') == 'process_started']
    base.require(launches == ['collector', 'runner'], 'Cont2 launched unexpected work')
    base.require(base.read_json(root / B / 'measurement-metadata.json').get('source_sha256') == source,
                 'Cont2 metadata source snapshot mismatch')
    base.verify_capture_closed(root / B / 'meter', root, strict=True)
    base.require(c.utc(result['finished']['utc']) <= c.utc(review['created_utc']),
                 'Deep continuation review predates closed cont2')
    return {**source, **{name: base.sha(root / name) for name in PROOFS}}


def preflight(root, functional_meter):
    c.verify_precedents(root)
    previous.verify_startup_failure(root)
    proofs = verify_B_failure(root)
    prepared, evidence = base.preflight(root, PREFIX, functional_meter)
    prepared = plan(root, prepared)
    base.require(all(proofs.get(k) == v for k, v in evidence['source_sha256'].items()),
                 'Original V2 sources changed')
    evidence['source_sha256'] = {**proofs,
        'continue_deepsleep_comparisons.py': base.sha(root / 'continue_deepsleep_comparisons.py')}
    evidence['continuation'] = prepared['continuation']
    prepared['continuation']['source_and_proof_sha256'] = evidence['source_sha256']
    base.verify_sources(root, evidence['source_sha256'])
    return prepared, evidence


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--label-prefix', default=PREFIX, choices=[PREFIX])
    parser.add_argument('--functional-meter-dir', type=Path, default=ROOT / 'meter-functional')
    group = parser.add_mutually_exclusive_group(required=True)
    for option in ('plan', 'check-only', 'execute'):
        group.add_argument('--' + option, action='store_true')
    args = parser.parse_args(argv)
    os.umask(0o077)
    base.require(not sys.flags.optimize, 'Optimized Python is not supported')
    c.configure()
    if args.plan:
        print(json.dumps(plan(ROOT), indent=2)); return 0
    with (ROOT / '.measure-variants.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        prepared, evidence = preflight(ROOT, args.functional_meter_dir.resolve())
        if args.check_only:
            print(json.dumps({'result': 'LOCAL_PREFLIGHT_PASS', 'hardware_opened': False,
                'profiles': len(prepared['profiles']), 'retained_A': c.A, 'B_result': 'FAIL',
                'source_sha256': evidence['source_sha256']})); return 0
        result = base.Supervisor(ROOT, prepared, evidence).run()
        print(json.dumps({'result': result['result'],
            'profiles_PASS': sum(r['result'] == 'PASS' for r in result['profiles']),
            'retained_A': c.A, 'B_result': 'FAIL', 'candidate_restored': result['candidate_restored'],
            'root_recovery_required': result['root_recovery_required']}))
        return 0 if result['result'] == 'PASS' else 1


if __name__ == '__main__':
    try:
        sys.exit(main())
    except Exception as exc:
        print(json.dumps({'result': 'ERROR_BEFORE_OR_OUTSIDE_SUPERVISED_RUN',
              'error_type': type(exc).__name__, 'reason': str(exc)}), file=sys.stderr)
        sys.exit(1)
